                }
            }
        }
    }
}

Q020핵심 기출큐·원형 큐의 연산

출처: 2. 2025년2회_정보처리기사실기 기출문제.pdf

다음 C 언어로 구현된 프로그램을 분석하여 그 실행 결과를 쓰시오. (단, 출력문의 출력 서식을 준수하시오.) (5점)
#include <stdio.h>
#define SIZE 3

typedef struct {
    int a[SIZE];
    int front;
    int rear;
} Queue;

void enq(Queue* q, int val) {
    q->a[q->rear] = val;
    q->rear = (q->rear + 1) % SIZE;
}

int deq(Queue* q) {
    int val = q->a[q->front];
    q->front = (q->front + 1) % SIZE;
    return val;
}

int main() {
    Queue q = {{0}, 0, 0};
    enq(&q, 1);
    enq(&q, 2);
    deq(&q);
    enq(&q, 3);
    printf("%d 그리고 %d", deq(&q), deq(&q));
    return 0;
}
정답·풀이 핵심 보기

정답: 3 그리고 2

복원본 정답은 “3 그리고 2”이다. enq(1), enq(2), deq(), enq(3) 뒤 큐에는 2·3이 남아 있다. 원문 해설은 printf의 오른쪽 deq를 먼저 호출하여 2를 뒤쪽 출력 자리에, 다음에 꺼낸 3을 앞쪽 자리에 넣는다. 주의: C는 함수 인수의 평가 순서를 고정하지 않으므로 왼쪽 deq를 먼저 평가하면 “2 그리고 3”도 가능하다. FIFO로 제거되는 순서 2→3과 출력 인수의 배치를 구별하고, 이 복원본 정답을 모든 C 환경의 보장된 결과로 외우지 않는다. 보충 근거: WG14 C11 위원회 초안 N1570 §6.5.2.2 문단 10·12.

    return (node.value if level % 2 == 1 else 0) + sum(
